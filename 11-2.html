四、核心代码 

.zhang_perspective {
  perspective: 1000px;        /* 透视深度 */
}
.zhang_preserve-3d {
  transform-style: preserve-3d; /* 保留 3D 空间 */
}
.zhang_cube-face {
  transform: rotateY(90deg) translateZ(60px); /* 每个面定位 */
}
@keyframes zhang_spin-y {
  from { transform: rotateY(0deg); }
  to   { transform: rotateY(360deg); }
}
